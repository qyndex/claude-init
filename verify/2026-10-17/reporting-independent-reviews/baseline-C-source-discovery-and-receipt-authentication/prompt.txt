36: def load_receipt_trust(path, repository, *, mode):
37:     """Parse separately reviewed trust data; callers must explicitly select a mode.
38: 
39:     This parser alone grants no receipt authority. Authentication integration must
40:     independently bind immutable GitHub source bytes and the selected profile.
41:     """
42:     require(isinstance(mode, str) and mode in {'profile', 'legacy', 'report'}, 'explicit receipt trust mode required')
43:     raw = Path(path).read_bytes()
44:     require(0 < len(raw) <= 262144, 'receipt trust document size exceeds bound')
45:     trust = json.loads(raw, object_pairs_hook=coordinator.contract.no_duplicate_keys)
46:     require(isinstance(trust, dict) and set(trust) == {'schema_version', 'repository', 'profiles', 'legacy_revisions'}, 'receipt trust schema differs')
47:     require(type(trust['schema_version']) is int and trust['schema_version'] == 1, 'receipt trust schema version differs')
48:     require(isinstance(repository, str) and re.fullmatch(r'[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+', repository) is not None and trust['repository'] == repository, 'receipt trust repository differs')
49: 
50:     def runtime(pins):
51:         require(isinstance(pins, dict) and frozenset(pins) in RUNTIME_FAMILIES, 'receipt trust runtime closure differs')
52:         require(all(coordinator.contract.digest(value, 64) for value in pins.values()), 'receipt trust runtime digest invalid')
53: 
54:     profiles = trust['profiles']
55:     require(isinstance(profiles, list) and len(profiles) <= 64, 'receipt trust profiles malformed')
56:     identities = set()

59:         identity = profile['id']
60:         require(isinstance(identity, str) and re.fullmatch(r'[a-z][a-z0-9-]{0,63}', identity) is not None, 'receipt trust profile identity invalid')

62:         identities.add(identity)

64:         require(coordinator.contract.digest(profile['policy_sha256'], 64), 'receipt trust policy digest invalid')

73:     elif mode == 'legacy':

136: def authenticate(api, policy, run_id, *, historical=False, trust_path=None, trust_mode='legacy'):
137:     require(type(run_id) is int and run_id > 0 and api.repo == policy['repository'], 'invalid receipt source')
138:     repo = policy['repository']; root = f'repos/{repo}'
139:     require(type(policy['receipts']['workflow_id']) is int and policy['receipts']['workflow_id'] > 0, 'receipt workflow not configured')
140:     run = api.get(f'{root}/actions/runs/{run_id}')
141:     require(type(run['id']) is int and run['id'] == run_id and type(run['run_attempt']) is int and run['run_attempt'] > 0, 'wrong run identity')
142:     require(run['repository']['full_name'] == repo and type(run['workflow_id']) is int and run['workflow_id'] == policy['receipts']['workflow_id'], 'foreign receipt producer')
143:     require(run['event'] in {'workflow_run', 'workflow_dispatch'} and run['head_branch'] == policy['base_branch'], 'unprotected receipt event/branch')
144:     require(run['status'] == 'completed' and run['conclusion'] == 'success', 'coordinator did not succeed')
145:     require(run.get('path') == RUNTIME_PATHS[0], 'receipt workflow path differs')
146:     require(coordinator.contract.digest(run.get('head_sha'), 40), 'receipt source SHA invalid')
147:     source, observed, trust, selected = authenticate_source(api, policy, run, trust_path=trust_path, trust_mode=trust_mode)
148:     document, artifact = receipt_artifact(api, run)
149:     expected = {'schema_version', 'repository', 'pull_request', 'head_sha', 'base_sha', 'spec', 'spec_sha256',
150:                 'policy_sha256', 'verifier_run', 'reviewer_run', 'eligible', 'merged', 'merge_sha', 'task_ids'}
151:     require(set(document) == expected and document['schema_version'] == 1, 'malformed delivery receipt')
152:     require(source.get('enabled') is True and source.get('merge_enabled') is True, 'receipt policy did not authorize merge')
153:     require(document['repository'] == repo == source['repository'] and document['eligible'] is True and document['merged'] is True, 'receipt is not a factory delivery')
154:     require(type(document['pull_request']) is int and document['pull_request'] > 0, 'invalid receipt PR')
155:     for name in ('head_sha', 'base_sha', 'merge_sha'):
156:         require(coordinator.contract.digest(document[name], 40), 'invalid candidate digest')
157:     require(all(type(document[k]) is int and document[k] > 0 for k in ('verifier_run', 'reviewer_run')) and document['verifier_run'] != document['reviewer_run'], 'missing independent proof identity')
158:     require(document['policy_sha256'] == coordinator.policy_hash(source), 'receipt policy differs')
159:     approved = source['specs'][document['spec']]
160:     current = approved if historical else policy['specs'][document['spec']]
161:     require(document['spec_sha256'] == approved['sha256'] == current['sha256'], 'superseded spec receipt')
162:     require(hashlib.sha256(api.source(approved['path'], run['head_sha'])).hexdigest() == document['spec_sha256'], 'receipt spec bytes differ')
163:     # Multiple task deliveries need explicit per-task acceptance mapping; never infer all spec tasks.
164:     tasks = document['task_ids']
165:     require(isinstance(tasks, list) and len(tasks) == 1 and tasks == approved.get('task_ids') == current.get('task_ids'), 'unapproved/ambiguous task delivery mapping')
166:     require(all(isinstance(t, str) and re.fullmatch(r'T-[1-9][0-9]*', t) for t in tasks), 'invalid delivery task')
167:     pr = api.get(f'{root}/pulls/{document["pull_request"]}')
168:     require(pr['number'] == document['pull_request'] and pr['merged'] is True and pr['merge_commit_sha'] == document['merge_sha'], 'actual PR merge differs')
169:     require(pr['head']['sha'] == document['head_sha'] and pr['head']['repo']['full_name'] == repo, 'actual candidate differs')
170:     require(pr['base']['repo']['full_name'] == repo and pr['base']['ref'] == policy['base_branch'] == source['base_branch'], 'actual merge target differs')
171:     delivery = {'repository': repo, 'pr': document['pull_request'], 'merge_sha': document['merge_sha']}
172:     encoded = json.dumps(delivery, sort_keys=True)
173:     provenance_facts = {'run_id': run_id, 'attempt': run['run_attempt'], 'workflow_id': run['workflow_id'],
174:                         'source_sha': run['head_sha'], 'artifact_id': artifact['id'], 'artifact_digest': artifact['digest'],
175:                         'receipt': document}
176:     if trust is not None:
177:         provenance_facts['execution'] = {'repository_id': run['repository']['id'], 'repository': repo, 'event': run['event'], 'branch': run['head_branch'], 'path': run['path'], 'status': run['status'], 'conclusion': run['conclusion']}
178:         provenance_facts['trust'] = {'mode': trust_mode, 'profile_id': selected['id'] if selected else None,
179:                                      'document_sha256': coordinator.policy_hash(trust),
180:                                      'policy_sha256': coordinator.policy_hash(source), 'runtime': observed}
181:     provenance = json.dumps(provenance_facts, sort_keys=True)
182:     return {'tasks': tasks, 'revision': document['spec_sha256'], 'delivery': encoded,
183:             'provenance': provenance, 'run_id': run_id, 'merge_sha': document['merge_sha']}

204: def ingest(api, policy, db_path, run_id, *, trust_path=None, trust_mode='legacy'):
205:     proof = authenticate(api, policy, run_id, trust_path=trust_path, trust_mode=trust_mode)
206:     tasks = proof['tasks']; encoded = proof['delivery']; provenance = proof['provenance']
207:     store = ledger.runtime.Store(db_path)
208:     with store.transaction() as db:
209:         db.execute('CREATE TABLE IF NOT EXISTS delivery_receipts(task TEXT PRIMARY KEY, revision TEXT NOT NULL, receipt TEXT NOT NULL)')
210:         db.execute('CREATE TABLE IF NOT EXISTS delivery_provenance(task TEXT PRIMARY KEY, provenance TEXT NOT NULL)')
211:         for task in tasks:
212:             row = store.row(db, task)
213:             require(row['revision'] == proof['revision'] and row['state'] != 'cancelled', 'runtime task authority differs')
214:             previous = db.execute('SELECT * FROM delivery_receipts WHERE task=?', (task,)).fetchone()
215:             old_proof = db.execute('SELECT * FROM delivery_provenance WHERE task=?', (task,)).fetchone()
216:             require(previous is None or (previous['revision'] == row['revision'] and previous['receipt'] == encoded and old_proof is not None and old_proof['provenance'] == provenance), 'conflicting delivery/provenance')
217:             db.execute('INSERT OR IGNORE INTO delivery_receipts VALUES(?,?,?)', (task, row['revision'], encoded))
218:             db.execute('INSERT OR IGNORE INTO delivery_provenance VALUES(?,?)', (task, provenance))
219:             if row['state'] != 'merged':
220:                 db.execute("UPDATE tasks SET state='merged',owner=NULL,expires=NULL,fence=fence+1 WHERE id=?", (task,))
221:     return {'completed_tasks': tasks, 'run_id': run_id, 'merge_sha': proof['merge_sha']}

224: def project(api, policy, db_path, *, receipt_mode='legacy', state_api=None, archive_config=None, selection=None):
225:     """Rebuild reporting-only proof tables; never create or transition runtime tasks."""
226:     require(type(receipt_mode) is str and receipt_mode in ('legacy', 'durable'), 'explicit reporting receipt mode required')
227:     if receipt_mode == 'durable':
228:         require(state_api is not None and type(archive_config) is dict and
229:                 set(archive_config)=={'approval_path','trust_path','store','trust_mode'} and
230:                 type(selection) is dict and set(selection)=={'interval_end','history_head','campaign'},
231:                 'protected durable reporting selection required')
232:         spec = importlib.util.spec_from_file_location('durable_reporting_receipts', Path(__file__).with_name('factory-reporting-receipts.py'))
233:         retained = importlib.util.module_from_spec(spec); spec.loader.exec_module(retained)
234:         try:
235:             return retained.project_reporting(api, state_api, policy, db_path, archive_config['trust_path'],
236:                         archive_config['store'], archive_config['approval_path'], retained.archive.archive_stream(api.repo),
237:                         interval_end=selection['interval_end'], history_head=selection['history_head'],
238:                         campaign=selection['campaign'], interval_scoped=True)
239:         except retained.ProjectionPending:
240:             return {'coverage_complete':False,'all_history_complete':False,'task_credit':False,
241:                     'gaps':['retained-reporting-projection-pending']}
242:     require(api.repo == policy['repository'], 'foreign reporting receipt policy')
243:     configured = policy['receipts']
244:     workflow = configured['workflow_id']
245:     if workflow is None:
246:         require(configured['trusted_revisions'] == {}, 'receipt workflow missing for approved revisions')
247:         proofs = []
248:     else:
249:         require(type(workflow) is int and workflow > 0, 'receipt workflow not configured')
250:         pages = api.request(f'repos/{api.repo}/actions/workflows/{workflow}/runs?status=success&per_page=100', paginate=True)
251:         require(isinstance(pages, list), 'receipt run pages malformed')
252:         runs = [run for page in pages for run in page['workflow_runs']]
253:         require(len(runs) <= 1000, 'receipt history exceeds projection bound')
254:         require(len({run['id'] for run in runs}) == len(runs), 'duplicate receipt run identity')
255:         proofs = []
256:         for run in runs:
257:             if run['head_sha'] not in configured['trusted_revisions']:
258:                 continue  # Unapproved sources can never become authenticated evidence.
259:             # A successful coordinator may deliberately decline a merge. Its artifact
260:             # is inspected as data only; only true delivery claims enter authentication.
261:             document, _ = receipt_artifact(api, run)
262:             require(type(document.get('merged')) is bool, 'receipt delivery outcome missing')
263:             if document['merged']:
264:                 proofs.append(authenticate(api, policy, run['id'], historical=True))
265:     # Validate the entire observation before atomically replacing the ephemeral tables.
266:     by_task = {}
267:     for proof in proofs:
268:         for task in proof['tasks']:
269:             require(task not in by_task or by_task[task] == proof, 'conflicting reporting delivery proof')
270:             by_task[task] = proof
271:     with sqlite3.connect(db_path) as db:
272:         db.execute('CREATE TABLE IF NOT EXISTS delivery_receipts(task TEXT PRIMARY KEY, revision TEXT NOT NULL, receipt TEXT NOT NULL)')
273:         db.execute('CREATE TABLE IF NOT EXISTS delivery_provenance(task TEXT PRIMARY KEY, provenance TEXT NOT NULL)')
274:         db.execute('DELETE FROM delivery_receipts')
275:         db.execute('DELETE FROM delivery_provenance')
276:         for task, proof in sorted(by_task.items()):
277:             db.execute('INSERT INTO delivery_receipts VALUES(?,?,?)', (task, proof['revision'], proof['delivery']))
278:             db.execute('INSERT INTO delivery_provenance VALUES(?,?)', (task, proof['provenance']))
279:     return len(by_task)
